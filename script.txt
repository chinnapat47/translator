
# ==========================================
# Auto File Organizer for Windows
# Real-time Downloads Organizer
# Original files are NEVER deleted
# ==========================================

$ErrorActionPreference = "Continue"

# Download folder
$SourceFolder = Join-Path $env:USERPROFILE "Downloads"

# Log file
$LogFolder = Join-Path $env:LOCALAPPDATA "AutoFileOrganizer"
$LogFile = Join-Path $LogFolder "Organizer.log"

New-Item -ItemType Directory -Path $LogFolder -Force | Out-Null

if (!(Test-Path -LiteralPath $SourceFolder)) {
    throw "Downloads folder not found: $SourceFolder"
}

# File categories
$Categories = @{
    "Documents" = @(
        ".pdf", ".doc", ".docx", ".xls", ".xlsx",
        ".ppt", ".pptx", ".txt", ".csv", ".rtf"
    )

    "Images" = @(
        ".jpg", ".jpeg", ".png", ".gif", ".bmp",
        ".webp", ".svg", ".ico", ".tiff"
    )

    "Videos" = @(
        ".mp4", ".mkv", ".mov", ".avi", ".wmv",
        ".flv", ".webm", ".m4v"
    )

    "Music" = @(
        ".mp3", ".wav", ".flac", ".aac",
        ".m4a", ".ogg", ".wma"
    )

    "Archives" = @(
        ".zip", ".rar", ".7z", ".tar",
        ".gz", ".bz2"
    )

    "Programs" = @(
        ".exe", ".msi", ".msix", ".appx"
    )

    "Code" = @(
        ".cs", ".vb", ".py", ".js", ".ts",
        ".html", ".css", ".sql", ".json",
        ".xml", ".php", ".java", ".cpp",
        ".c", ".h", ".ps1", ".bat", ".cmd"
    )
}

# Temporary download extensions
$IgnoredExtensions = @(
    ".crdownload", ".part", ".tmp",
    ".download", ".partial"
)

function Write-Log {
    param([string]$Message)

    $Time = Get-Date -Format "yyyy-MM-dd HH:mm:ss"
    Add-Content -LiteralPath $LogFile `
        -Value "[$Time] $Message" -Encoding UTF8
}

function Get-Category {
    param([string]$Extension)

    foreach ($Category in $Categories.Keys) {
        if ($Categories[$Category] -contains $Extension) {
            return $Category
        }
    }

    return "Others"
}

function Wait-FileReady {
    param([string]$Path)

    # Wait until file size stops changing
    $LastSize = -1
    $StableCount = 0

    for ($i = 0; $i -lt 120; $i++) {
        if (!(Test-Path -LiteralPath $Path -PathType Leaf)) {
            return $false
        }

        try {
            $File = Get-Item -LiteralPath $Path -ErrorAction Stop

            if ($File.Length -eq $LastSize) {
                $StableCount++
            }
            else {
                $StableCount = 0
                $LastSize = $File.Length
            }

            if ($StableCount -ge 3) {
                # Verify that no other process is writing
                $Stream = [System.IO.File]::Open(
                    $Path,
                    [System.IO.FileMode]::Open,
                    [System.IO.FileAccess]::Read,
                    [System.IO.FileShare]::None
                )
                $Stream.Dispose()
                return $true
            }
        }
        catch {
            $StableCount = 0
        }

        Start-Sleep -Seconds 1
    }

    return $false
}

function Get-UniqueDestination {
    param(
        [string]$Folder,
        [string]$FileName,
        [long]$FileSize
    )

    $Destination = Join-Path $Folder $FileName

    if (!(Test-Path -LiteralPath $Destination)) {
        return $Destination
    }

    # Skip if the same name and size already exist
    if ((Get-Item -LiteralPath $Destination).Length -eq $FileSize) {
        return $null
    }

    $Name = [System.IO.Path]::GetFileNameWithoutExtension($FileName)
    $Ext = [System.IO.Path]::GetExtension($FileName)

    $Counter = 1

    do {
        $NewName = "${Name}_$Counter$Ext"
        $Destination = Join-Path $Folder $NewName
        $Counter++
    }
    while (Test-Path -LiteralPath $Destination)

    return $Destination
}

function Organize-File {
    param([string]$Path)

    try {
        if (!(Test-Path -LiteralPath $Path -PathType Leaf)) {
            return
        }

        $File = Get-Item -LiteralPath $Path -ErrorAction Stop

        # Ignore temporary download files
        if ($IgnoredExtensions -contains $File.Extension.ToLower()) {
            return
        }

        # Only organize files directly in Downloads
        if ($File.DirectoryName -ne $SourceFolder) {
            return
        }

        if (!(Wait-FileReady -Path $Path)) {
            Write-Log "SKIPPED (file not ready): $Path"
            return
        }

        $File = Get-Item -LiteralPath $Path
        $Category = Get-Category -Extension $File.Extension.ToLower()

        $TargetFolder = Join-Path $SourceFolder $Category

        New-Item -ItemType Directory `
            -Path $TargetFolder -Force | Out-Null

        $Destination = Get-UniqueDestination `
            -Folder $TargetFolder `
            -FileName $File.Name `
            -FileSize $File.Length

        if ($null -eq $Destination) {
            return
        }

        # COPY, NOT MOVE. Original file remains untouched.
        Copy-Item -LiteralPath $Path `
            -Destination $Destination -ErrorAction Stop

        # Verify copied file
        $Copied = Get-Item -LiteralPath $Destination

        if ($Copied.Length -ne $File.Length) {
            Remove-Item -LiteralPath $Destination -ErrorAction SilentlyContinue
            throw "Copy verification failed"
        }

        Write-Log "COPIED: $($File.Name) -> $Category"

        Write-Host "Organized: $($File.Name) -> $Category" `
            -ForegroundColor Green
    }
    catch {
        Write-Log "ERROR: $Path | $($_.Exception.Message)"
        Write-Host "Error: $($_.Exception.Message)" `
            -ForegroundColor Red
    }
}

# ==========================================
# INITIAL SCAN: Organize existing files
# ==========================================

Write-Host "Scanning existing Downloads..." `
    -ForegroundColor Cyan

Get-ChildItem -LiteralPath $SourceFolder -File -Force |
    ForEach-Object {
        Organize-File -Path $_.FullName
    }

Write-Host "Initial scan completed." -ForegroundColor Green

# ==========================================
# REAL-TIME FILE WATCHER
# ==========================================

$Watcher = New-Object System.IO.FileSystemWatcher

$Watcher.Path = $SourceFolder
$Watcher.Filter = "*"
$Watcher.IncludeSubdirectories = $false
$Watcher.NotifyFilter = [IO.NotifyFilters]::FileName `
    -bor [IO.NotifyFilters]::Size `
    -bor [IO.NotifyFilters]::LastWrite

$Watcher.EnableRaisingEvents = $true

Write-Host "==================================" -ForegroundColor Cyan
Write-Host " AUTO FILE ORGANIZER IS RUNNING" -ForegroundColor Green
Write-Host " Watching: $SourceFolder" -ForegroundColor Cyan
Write-Host " Mode: REAL-TIME COPY" -ForegroundColor Yellow
Write-Host " Original files are preserved." -ForegroundColor Green
Write-Host " Press CTRL+C to stop." -ForegroundColor Gray
Write-Host "==================================" -ForegroundColor Cyan

try {
    $LastRescan = Get-Date

    while ($true) {

        # Wait for filesystem changes
        $Result = $Watcher.WaitForChanged(
            [IO.WatcherChangeTypes]::All,
            1000
        )

        if (!$Result.TimedOut -and $Result.Name) {
            $ChangedPath = Join-Path $SourceFolder $Result.Name

            Start-Sleep -Milliseconds 500
            Organize-File -Path $ChangedPath
        }

        # Periodic recovery scan every 30 seconds
        if (((Get-Date) - $LastRescan).TotalSeconds -ge 30) {

            Get-ChildItem -LiteralPath $SourceFolder -File -Force |
                ForEach-Object {
                    Organize-File -Path $_.FullName
                }

            $LastRescan = Get-Date
        }
    }
}
finally {
    $Watcher.EnableRaisingEvents = $false
    $Watcher.Dispose()
}
